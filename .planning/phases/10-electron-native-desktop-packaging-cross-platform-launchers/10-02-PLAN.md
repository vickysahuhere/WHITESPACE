  <files>package.json</files>
  <read_first>package.json</read_first>
  <action>
    Update `package.json`:
    - Add `"main": "electron/main.cjs"`.
    - Add scripts:
      - `"electron:start": "electron electron/main.cjs"`.
      - `"electron:dev": "concurrently -k \"vite\" \"wait-on http://localhost:5173 && cross-env NODE_ENV=development electron electron/main.cjs\""`.
      - `"electron:build": "npm run build && electron-builder --win --x64"`.
      - `"electron:pack": "npm run build && electron-builder --dir"`.
    - Add `"build"` block for `electron-builder`:
      - `appId: "com.whitecollars.whitespace"`.
      - `productName: "WhiteCollarsOS"`.
      - `copyright: "Copyright (C) 2026 White Collars"`.
      - `directories: { "output": "dist-electron" }`.
      - `files: ["dist/**/*", "electron/**/*"]`.
      - `win: { "target": ["portable", "nsis"], "artifactName": "${productName}_v${version}_${arch}.${ext}" }`.
      - `linux: { "target": ["AppImage", "tar.gz"], "category": "Development" }`.
  </action>
  <verify>npm run build</verify>
  <acceptance_criteria>
    - package.json includes main entry and electron scripts

  </acceptance_criteria>

<task type="auto">
  <name>Task 2: Create LAUNCH_WINDOWS_APP.bat, BUILD_EXE_INSTALLER.bat, and launch_fedora.sh</name>
  <files>LAUNCH_WINDOWS_APP.bat, BUILD_EXE_INSTALLER.bat, launch_fedora.sh</files>
  <read_first>package.json</read_first>
  <action>
    Create `LAUNCH_WINDOWS_APP.bat`:
      - Checks for Node.js and npm in PATH.
      - Checks node_modules; if missing, runs `npm install`.
      - Prints monochrome ASCII header: `WHITE COLLARS (WHITESPACE) // AUTONOMOUS WORKSPACE OS`.
      - Launches `npm run electron:dev`.

    Create `BUILD_EXE_INSTALLER.bat`:
      - Verifies Node.js.
      - Prints packaging banner: `COMPILING STANDALONE WINDOWS EXECUTABLE (.EXE)...`.
      - Executes `npm run electron:build`.
      - Displays location of compiled binary in `dist-electron/`.

    Create `launch_fedora.sh`:
      - POSIX shell script with executable permission.
      - Handles Wayland/X11 auto-detection (`--ozone-platform-hint=auto`).
      - Runs `npm run electron:dev`.
  </action>
  <verify>test batch scripts and shell script exist</verify>
  <acceptance_criteria>
    - LAUNCH_WINDOWS_APP.bat launches electron:dev
    - BUILD_EXE_INSTALLER.bat compiles executable
    - launch_fedora.sh supports Linux
  </acceptance_criteria>
